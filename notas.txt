CREATE COMPUTE MODULE Country_Info_Compute
    CREATE FUNCTION Main() RETURNS BOOLEAN
    BEGIN
        -- Extraer datos de entrada y guardarlos en Environment
        SET Environment.RequestEntrada = InputRoot.JSON.Data;
        
        -- Validar que los datos de entrada tengan formato correcto
        IF EXISTS(InputRoot.JSON.Data.requestType[]) AND 
           EXISTS(InputRoot.JSON.Data.parameters[]) THEN
            SET Environment.ValidationStatus = 'VALID';
        ELSE
            SET Environment.ValidationStatus = 'INVALID';
            -- Crear un mensaje de error
            SET OutputRoot.JSON.Data.responseInfo.status = 'error';
            SET OutputRoot.JSON.Data.responseInfo.message = 'Invalid request format';
            RETURN FALSE;
        END IF;
        
        RETURN TRUE;
    END;
END MODULE;

2. Adaptación para mapear la solicitud JSON a SOAP:
CREATE COMPUTE MODULE RequestToSOAPMapper
    CREATE FUNCTION Main() RETURNS BOOLEAN
    BEGIN
        -- Determinar qué operación SOAP invocar basado en el requestType y searchType
        DECLARE requestType CHARACTER Environment.RequestEntrada.requestType;
        DECLARE searchType CHARACTER Environment.RequestEntrada.parameters.searchType;
        
        -- Crear estructura SOAP para la salida
        CREATE LASTCHILD OF OutputRoot DOMAIN 'XMLNSC';
        
        -- Mapear diferentes tipos de solicitudes a operaciones SOAP específicas
        IF requestType = 'countryInfo' THEN
            IF searchType = 'byCode' THEN
                -- Mapear a FullCountryInfo
                SET Environment.SOAPOperation = 'FullCountryInfo';
                
                -- Crear el mensaje SOAP para FullCountryInfo
                SET OutputRoot.XMLNSC.(SOAP.Envelope).(SOAP.Body).{'ns:FullCountryInfo'} = '';
                DECLARE soapBody REFERENCE TO OutputRoot.XMLNSC.(SOAP.Envelope).(SOAP.Body).{'ns:FullCountryInfo'};
                
                -- Establecer el código del país
                SET soapBody.{'ns:sCountryISOCode'} = Environment.RequestEntrada.parameters.countryCode;
                
                -- Configurar el namespace
                SET OutputRoot.XMLNSC.(SOAP.Envelope).(XMLNSC.NamespaceDecl)xmlns:ns = 'http://www.oorsprong.org/websamples.countryinfo';
            ELSEIF searchType = 'byName' THEN
                -- Mapear a CountryISOCode
                SET Environment.SOAPOperation = 'CountryISOCode';
                
                -- Crear el mensaje SOAP para CountryISOCode
                SET OutputRoot.XMLNSC.(SOAP.Envelope).(SOAP.Body).{'ns:CountryISOCode'} = '';
                DECLARE soapBody REFERENCE TO OutputRoot.XMLNSC.(SOAP.Envelope).(SOAP.Body).{'ns:CountryISOCode'};
                
                -- Establecer el nombre del país
                SET soapBody.{'ns:sCountryName'} = Environment.RequestEntrada.parameters.countryName;
                
                -- Configurar el namespace
                SET OutputRoot.XMLNSC.(SOAP.Envelope).(XMLNSC.NamespaceDecl)xmlns:ns = 'http://www.oorsprong.org/websamples.countryinfo';
            END IF;
        ELSEIF requestType = 'listCountries' THEN
            -- Mapear a ListOfCountryNamesByCode o ListOfCountryNamesByName
            IF searchType = 'byCode' THEN
                SET Environment.SOAPOperation = 'ListOfCountryNamesByCode';
                
                -- Crear el mensaje SOAP para ListOfCountryNamesByCode
                SET OutputRoot.XMLNSC.(SOAP.Envelope).(SOAP.Body).{'ns:ListOfCountryNamesByCode'} = '';
                
                -- Configurar el namespace
                SET OutputRoot.XMLNSC.(SOAP.Envelope).(XMLNSC.NamespaceDecl)xmlns:ns = 'http://www.oorsprong.org/websamples.countryinfo';
            ELSE
                SET Environment.SOAPOperation = 'ListOfCountryNamesByName';
                
                -- Crear el mensaje SOAP para ListOfCountryNamesByName
                SET OutputRoot.XMLNSC.(SOAP.Envelope).(SOAP.Body).{'ns:ListOfCountryNamesByName'} = '';
                
                -- Configurar el namespace
                SET OutputRoot.XMLNSC.(SOAP.Envelope).(XMLNSC.NamespaceDecl)xmlns:ns = 'http://www.oorsprong.org/websamples.countryinfo';
            END IF;
        END IF;
        
        -- Configurar el endpoint SOAP
        SET OutputLocalEnvironment.Destination.SOAP.Request.Operation = Environment.SOAPOperation;
        SET OutputLocalEnvironment.Destination.SOAP.Request.Transport.URI = 'http://webservices.oorsprong.org/websamples.countryinfo/CountryInfoService.wso';
        
        RETURN TRUE;
    END;
END MODULE;

3. Adaptación para transformar la respuesta SOAP a JSON:
CREATE COMPUTE MODULE SOAPResponseToJSON
    CREATE FUNCTION Main() RETURNS BOOLEAN
    BEGIN
        -- Crear estructura JSON para la salida
        CREATE LASTCHILD OF OutputRoot DOMAIN 'JSON';
        CREATE FIELD OutputRoot.JSON.Data;
        
        -- Establecer información básica de respuesta
        SET OutputRoot.JSON.Data.responseInfo.status = 'success';
        SET OutputRoot.JSON.Data.responseInfo.timestamp = CURRENT_TIMESTAMP;
        SET OutputRoot.JSON.Data.responseInfo.requestType = Environment.RequestEntrada.requestType;
        
        -- Procesar diferentes tipos de respuestas SOAP
        DECLARE soapOperation CHARACTER Environment.SOAPOperation;
        
        IF soapOperation = 'FullCountryInfo' THEN
            -- Procesar respuesta de FullCountryInfo
            DECLARE soapResp REFERENCE TO InputRoot.XMLNSC.(SOAP.Body).{'*:FullCountryInfoResponse'}.{'*:FullCountryInfoResult'};
            
            -- Mapear datos básicos
            SET OutputRoot.JSON.Data.countryData.basicInfo.name = soapResp.{'*:sName'};
            SET OutputRoot.JSON.Data.countryData.basicInfo.code = soapResp.{'*:sISOCode'};
            SET OutputRoot.JSON.Data.countryData.basicInfo.capital = soapResp.{'*:sCapitalCity'};
            
            -- Mapear detalles
            SET OutputRoot.JSON.Data.countryData.details.continent.code = soapResp.{'*:sContinentCode'};
            SET OutputRoot.JSON.Data.countryData.details.currency.code = soapResp.{'*:sCurrencyISOCode'};
            SET OutputRoot.JSON.Data.countryData.details.phoneCode = soapResp.{'*:sPhoneCode'};
            SET OutputRoot.JSON.Data.countryData.details.flagUrl = soapResp.{'*:sCountryFlag'};
            
            -- Mapear idiomas
            DECLARE i INTEGER 1;
            DECLARE langRef REFERENCE TO soapResp.{'*:Languages'}.{'*:tLanguage'}[i];
            DECLARE j INTEGER 1;
            
            WHILE LASTMOVE(langRef) DO
                SET OutputRoot.JSON.Data.countryData.languages[j].name = langRef.{'*:sName'};
                SET OutputRoot.JSON.Data.countryData.languages[j].code = langRef.{'*:sISOCode'};
                SET OutputRoot.JSON.Data.countryData.languages[j].primary = (j = 1); -- Primer idioma es primario
                
                SET i = i + 1;
                SET j = j + 1;
                MOVE langRef TO soapResp.{'*:Languages'}.{'*:tLanguage'}[i];
            END WHILE;
            
        ELSEIF soapOperation = 'CountryISOCode' THEN
            -- Procesar respuesta de CountryISOCode
            DECLARE isoCode CHARACTER InputRoot.XMLNSC.(SOAP.Body).{'*:CountryISOCodeResponse'}.{'*:CountryISOCodeResult'};
            
            SET OutputRoot.JSON.Data.countryData.basicInfo.code = isoCode;
            
        ELSEIF soapOperation = 'ListOfCountryNamesByCode' OR soapOperation = 'ListOfCountryNamesByName' THEN
            -- Procesar respuesta de ListOfCountryNamesByCode o ListOfCountryNamesByName
            DECLARE responseField CHARACTER;
            IF soapOperation = 'ListOfCountryNamesByCode' THEN
                SET responseField = '*:ListOfCountryNamesByCodeResponse';
            ELSE
                SET responseField = '*:ListOfCountryNamesByNameResponse';
            END IF;
            
            DECLARE resultField CHARACTER;
            IF soapOperation = 'ListOfCountryNamesByCode' THEN
                SET resultField = '*:ListOfCountryNamesByCodeResult';
            ELSE
                SET resultField = '*:ListOfCountryNamesByNameResult';
            END IF;
            
            DECLARE countriesRef REFERENCE TO InputRoot.XMLNSC.(SOAP.Body).{responseField}.{resultField}.{'*:tCountryCodeAndName'};
            DECLARE i INTEGER 1;
            DECLARE j INTEGER 1;
            
            -- Crear array para países
            CREATE FIELD OutputRoot.JSON.Data.countries IDENTITY JSON.Array;
            
            WHILE LASTMOVE(countriesRef[i]) DO
                SET OutputRoot.JSON.Data.countries.Item[j].name = countriesRef[i].{'*:sName'};
                SET OutputRoot.JSON.Data.countries.Item[j].code = countriesRef[i].{'*:sISOCode'};
                
                SET i = i + 1;
                SET j = j + 1;
            END WHILE;
        END IF;
        
        RETURN TRUE;
    END;
END MODULE;

CREATE FILTER MODULE country_info_Filter

               AND Environment.RequestEntrada.parameters.includeDetails = TRUE;
